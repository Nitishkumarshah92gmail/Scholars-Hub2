//     implementation "com.google.firebase:firebase-messaging:23.4.1"   // ← ADD (enables FCM)
//     // ... other Capacitor plugin dependencies added by cap sync ...
// }
//
// apply plugin: 'com.google.gms.google-services'   // ← ADD THIS (must be last)

// ─────────────────────────────────────────────────────────────────────────────
